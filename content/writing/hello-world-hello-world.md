int maxSubarraySum(const std::vector<int>& nums, int k) {
    int n = nums.size();
    if (n < k) return 0;

    // Calculate sum of the first window
    int window_sum = 0;
    for (int i = 0; i < k; ++i) {
        window_sum += nums[i];
    }

    int max_sum = window_sum;

    // Slide the window from index k to the end of the array
    for (int i = k; i < n; ++i) {
        window_sum += nums[i] - nums[i - k]; // Add next element, subtract leftmost element
        max_sum = std::max(max_sum, window_sum);
    }

    return max_sum;
}
